cmake_minimum_required(VERSION 3.8)
project(zhishu_robot_workobject_calibration)
find_package(ament_cmake REQUIRED)
install(PROGRAMS scripts/workobject_sampler.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/workobject_math.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(workobject_math test/test_workobject_math.py)
  ament_add_pytest_test(probe_selection test/test_probe_selection.py TIMEOUT 120)
endif()
ament_package()
